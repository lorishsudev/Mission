⚙️ 核心驗證邏輯速查（Edge Function 開發參考）

GPS_PHOTO 驗證流程

  1. App 上傳原圖 → challenge-uploads/{userId}/{challengeId}/{timestamp}.jpg
  2. Edge Function 觸發：下載照片 → 用 exifr 套件解析 GPSLatitude、GPSLongitude
  3. Haversine 公式計算距離 d（公尺），與 verification_criteria.radius_meters 比較
  4. d ≤ radius_meters → is_passed = true
// Haversine 公式（TypeScript）
function haversine(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const R = 6_371_000; // 地球半徑（公尺）
  const toRad = (deg: number) => (deg * Math.PI) / 180;
  const dLat = toRad(lat2 - lat1);
  const dLon = toRad(lon2 - lon1);
  const a = Math.sin(dLat / 2) ** 2 +
            Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLon / 2) ** 2;
  return 2 * R * Math.asin(Math.sqrt(a));
}
